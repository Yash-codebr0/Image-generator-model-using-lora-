## LORA adapter
#training

python -m src.fine_tune.train_lora `
  --pretrained_model "runwayml/stable-diffusion-v1-5" `
  --image_dir data/images `
  --captions data/captions.jsonl `
  --output_dir outputs/my_lora `
  --resolution 512 `
  --train_batch_size 1 `
  --max_train_steps 100

#generate 
python -m src.fine_tune.generate `
  --base_model "runwayml/stable-diffusion-v1-5" `
  --lora outputs/my_lora `
  --prompt "a futuristic city at night, cinematic lighting, 8k wallpaper" `
  --height 512 `
  --width 768 `
  --upscale_factor 2 `
  --output outputs/generated_lora_hd.png
